## 1. Prisma schema

ไฟล์ `apps/api/prisma/schema.prisma` (`apps/api` เป็นที่เดียวที่ต่อ DB) — generator/datasource block ปรับตาม Prisma version ที่ pin ใน Phase 1 (Prisma 7 ย้าย connection URL ไป `prisma.config.ts`)

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ───────── Enums ─────────

enum UserRole {
  ADMIN
  SALES
}

enum LeadStage {
  NEW
  QUALIFIED
  PROPOSAL
  WON
  LOST
}

enum LeadSource {
  WEBSITE
  MANUAL
  LINE
}

enum ActivityType {
  NOTE
  CALL
  MEETING
  TASK
  STAGE_CHANGE // ระบบสร้าง: metadata = { from, to, lostReason? }
  AI_APPROVED  // ระบบสร้าง: metadata = { suggestionId, suggestionType }
  AI_REJECTED  // ระบบสร้าง: metadata = { suggestionId, suggestionType, reason? }
  SYSTEM       // lead ถูกสร้าง, เปลี่ยน owner, ผูก LINE
}

enum MessageDirection {
  INBOUND
  OUTBOUND
}

enum MessageChannel {
  LINE
  WEB_FORM
}

enum MessageStatus {
  RECEIVED // inbound
  QUEUED   // outbound: บันทึกแล้ว รอส่ง
  SENT
  FAILED   // outbound: ส่งไม่สำเร็จหลัง retry อัตโนมัติ → กด retry เองได้
}

enum AiSuggestionType {
  QUALIFICATION // summary + score + reasons → approve = เขียน Lead.summary / Lead.score
  NEXT_ACTION   // → approve = สร้าง Activity ประเภท TASK
  LINE_REPLY    // → approve = สร้าง Message OUTBOUND แล้วส่ง LINE
}

enum AiSuggestionStatus {
  PENDING
  APPROVED
  REJECTED
  SUPERSEDED // มี suggestion ใหม่ประเภทเดียวกันมาแทน → approve ไม่ได้แล้ว
}

enum AiSource {
  LLM
  FALLBACK // rule-based เมื่อ AI ล่ม / timeout / output ไม่ผ่าน zod / ไม่มี API key
}

enum WebhookEventStatus {
  RECEIVED
  PROCESSED
  FAILED
  IGNORED // event type ที่ยังไม่รองรับ
}

// ───────── Models ─────────

model User {
  id           String   @id @default(cuid())
  email        String   @unique
  name         String
  passwordHash String
  role         UserRole @default(SALES)
  isActive     Boolean  @default(true) // ไม่ลบ user จริง → FK ของ audit ไม่ขาด
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  ownedLeads           Lead[]         @relation("LeadOwner")
  activities           Activity[]
  sentMessages         Message[]
  requestedSuggestions AiSuggestion[] @relation("SuggestionRequester")
  reviewedSuggestions  AiSuggestion[] @relation("SuggestionReviewer")
}

model Company {
  id            String   @id @default(cuid())
  name          String
  domain        String?  @unique
  industry      String?
  employeeCount Int?
  createdAt     DateTime @default(now())
  updatedAt     DateTime @updatedAt

  contacts Contact[]
  leads    Lead[]

  @@index([name])
}

model Contact {
  id              String   @id @default(cuid())
  name            String
  email           String?  @unique // เก็บเป็น lowercase (zod transform)
  phone           String?
  jobTitle        String?
  lineUserId      String?  @unique // LINE user ↔ contact แบบ 1:1
  lineDisplayName String?
  companyId       String?
  company         Company? @relation(fields: [companyId], references: [id], onDelete: Restrict)
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt

  leads    Lead[]
  messages Message[]

  @@index([companyId])
  @@index([name])
}

model Lead {
  id             String     @id @default(cuid())
  title          String
  stage          LeadStage  @default(NEW)
  source         LeadSource
  value          Decimal?   @db.Decimal(12, 2) // มูลค่าดีล (THB)
  score          Int?       // 0–100 ค่าที่ "ยืนยันแล้ว" (มาจาก approve หรือคนแก้เอง)
  summary        String?    // สรุปที่ยืนยันแล้ว
  lostReason     String?
  stageChangedAt DateTime   @default(now())
  closedAt       DateTime?  // set เมื่อ WON / LOST
  contactId      String
  contact        Contact    @relation(fields: [contactId], references: [id], onDelete: Restrict)
  companyId      String?
  company        Company?   @relation(fields: [companyId], references: [id], onDelete: Restrict)
  ownerId        String?    // null = ยังไม่ assign (lead จาก LINE)
  owner          User?      @relation("LeadOwner", fields: [ownerId], references: [id], onDelete: SetNull)
  createdAt      DateTime   @default(now())
  updatedAt      DateTime   @updatedAt

  activities    Activity[]
  messages      Message[]
  aiSuggestions AiSuggestion[]

  @@index([stage, updatedAt])
  @@index([ownerId, stage])
  @@index([contactId])
}

model Activity {
  id          String       @id @default(cuid())
  leadId      String
  lead        Lead         @relation(fields: [leadId], references: [id], onDelete: Restrict)
  type        ActivityType
  body        String?
  metadata    Json?
  dueAt       DateTime?    // TASK เท่านั้น
  completedAt DateTime?    // TASK เท่านั้น — field เดียวที่แก้ได้หลังสร้าง
  actorId     String?      // null = ระบบ (เช่น LINE webhook)
  actor       User?        @relation(fields: [actorId], references: [id], onDelete: Restrict)
  createdAt   DateTime     @default(now())

  @@index([leadId, createdAt])
}

model Message {
  id             String           @id @default(cuid())
  leadId         String
  lead           Lead             @relation(fields: [leadId], references: [id], onDelete: Restrict)
  contactId      String
  contact        Contact          @relation(fields: [contactId], references: [id], onDelete: Restrict)
  direction      MessageDirection
  channel        MessageChannel
  status         MessageStatus
  text           String
  lineMessageId  String?          @unique // inbound: message id จาก LINE (กันซ้ำชั้นที่ 2)
  retryKey       String?          @unique @db.Uuid // outbound: X-Line-Retry-Key ใช้ค่าเดิมทุกครั้งที่ retry
  attempts       Int              @default(0)
  lastError      String?
  sentAt         DateTime?
  sentById       String?          // outbound: คนที่กดส่ง / กด approve
  sentBy         User?            @relation(fields: [sentById], references: [id], onDelete: Restrict)
  aiSuggestionId String?          @unique // outbound ที่มาจาก AI draft ที่ approve แล้ว
  aiSuggestion   AiSuggestion?    @relation(fields: [aiSuggestionId], references: [id], onDelete: Restrict)
  webhookEventId String?          @unique // inbound: event ต้นทาง
  webhookEvent   WebhookEvent?    @relation(fields: [webhookEventId], references: [id], onDelete: Restrict)
  createdAt      DateTime         @default(now())

  @@index([leadId, createdAt])
  @@index([status])
}

model AiSuggestion {
  id             String             @id @default(cuid())
  leadId         String
  lead           Lead               @relation(fields: [leadId], references: [id], onDelete: Restrict)
  type           AiSuggestionType
  status         AiSuggestionStatus @default(PENDING)
  payload        Json               // สิ่งที่ AI เสนอ (ผ่าน zod schema ตาม type แล้ว)
  finalPayload   Json?              // สิ่งที่คน approve จริง (อาจแก้ไข เช่นแก้ข้อความ reply)
  source         AiSource
  aiModel        String?            // เช่น "claude-sonnet-5"; null เมื่อ FALLBACK
  promptVersion  String             // เช่น "crm-copilot@1"
  fallbackReason String?            // timeout | provider_error | refusal | invalid_output | no_api_key
  latencyMs      Int?
  requestedById  String?            // null = ระบบสร้างเองจาก LINE inbound
  requestedBy    User?              @relation("SuggestionRequester", fields: [requestedById], references: [id], onDelete: Restrict)
  reviewedById   String?
  reviewedBy     User?              @relation("SuggestionReviewer", fields: [reviewedById], references: [id], onDelete: Restrict)
  reviewedAt     DateTime?
  rejectReason   String?
  createdAt      DateTime           @default(now())

  message Message? // outbound ที่เกิดจากการ approve LINE_REPLY

  @@index([leadId, type, status])
}

model WebhookEvent {
  id           String             @id @default(cuid())
  eventId      String             @unique // LINE webhookEventId → idempotency key
  type         String             // message | follow | unfollow | ...
  lineUserId   String?
  payload      Json               // event ดิบ — บันทึกหลัง verify signature ผ่านแล้วเท่านั้น
  isRedelivery Boolean            @default(false)
  status       WebhookEventStatus @default(RECEIVED)
  attempts     Int                @default(0)
  lastError    String?
  nextRetryAt  DateTime?
  processedAt  DateTime?
  receivedAt   DateTime           @default(now())

  message Message?

  @@index([status, nextRetryAt])
}
```

### Constraint ที่เขียนใน Prisma schema ไม่ได้ → raw SQL ใน migration แรก

- `Lead`: `CHECK (score IS NULL OR score BETWEEN 0 AND 100)`, `CHECK (value IS NULL OR value >= 0)`, `CHECK (stage <> 'LOST' OR "lostReason" IS NOT NULL)`
- `Lead` (เพิ่มตอน Phase 1): `CHECK (("stage" IN ('WON','LOST')) = ("closedAt" IS NOT NULL))`
- `AiSuggestion`: partial unique index `("leadId", type) WHERE status = 'PENDING'` → มี draft รออนุมัติได้ทีละ 1 ต่อประเภทต่อ lead แม้มี request พร้อมกัน — **อัปเดต Phase 1:** ประกาศใน schema ด้วย `@@unique([leadId, type], where: { status: "PENDING" })` (preview `partialIndexes`) ไม่ต้องใช้ raw SQL

### ทำไมต้องมี `AiSuggestion`

1. **เป็นกำแพงระหว่าง AI กับข้อมูลจริง** — output ของ AI ลงที่ตารางนี้เท่านั้น (สถานะ `PENDING`) ส่วน `Lead` / `Activity` / `Message` จะเปลี่ยนได้ผ่าน approve endpoint ที่คนกดเท่านั้น ตรงกับโจทย์ Part 2 ("Separate AI suggestions from confirmed database writes or outbound messages") และข้อห้ามใน `CLAUDE.md`
2. **Audit trail ครบ** — เก็บว่า AI เสนออะไร (`payload`), คนแก้เป็นอะไร (`finalPayload`), ใครอนุมัติเมื่อไร และ `Message.aiSuggestionId` ชี้กลับมาที่ draft ต้นทางได้ ตรงกับโจทย์ "approve a LINE reply without losing the audit trail"
3. **วัดคุณภาพ AI และ fallback ได้** — `source`, `fallbackReason`, `aiModel`, `promptVersion`, `latencyMs` ทำให้วัด fallback rate, approve rate และ edit rate ได้ ใช้เป็นข้อมูลสำหรับ eval, monitoring และหาจุดปรับปรุง (Spot Opportunities)
4. **กันการกระทำซ้ำ** — status ร่วมกับ partial unique index ทำให้ approve ซ้ำไม่ได้ (ไม่ส่ง LINE ซ้ำ) และ draft เก่าจะถูก `SUPERSEDED` เมื่อมีข้อความใหม่เข้ามา จึงไม่เผลอส่งคำตอบที่ล้าสมัย

### ทำไมต้องมี `WebhookEvent`

1. **Idempotency** — LINE ส่ง event ซ้ำได้ (redelivery หรือฝั่งเรา timeout) แต่ `eventId` เป็น unique จึงประมวลผลแค่ครั้งเดียว ไม่เกิด message หรือ lead ซ้ำ
2. **ตอบ LINE เร็วโดยไม่เสียข้อมูล** — บันทึก event ดิบก่อน ตอบ 200 ทันที แล้วค่อยประมวลผล (map contact/lead, เรียก AI) ถ้า DB หรือ AI ล่มระหว่างทาง event ยังอยู่ในสถานะ `FAILED` และ retry ได้ ตรงกับโจทย์ "persist the event" และ "retry/idempotency"
3. **Debug, replay และ monitoring** — มี payload ดิบให้ดูย้อนหลังหรือ replay หลังแก้ bug และจำนวน event ที่ `FAILED` ใช้เป็นสัญญาณ alert ได้
4. **แยกสิ่งที่ LINE ส่งมาออกจากข้อมูลธุรกิจ** — event ที่ยังไม่รองรับ (follow/unfollow/sticker) เก็บไว้ได้โดยไม่ต้องยัดลงตาราง `Message`

### กติกาของข้อมูล

- **Stage transition** (pure function `canTransition()` ใน `packages/shared` + unit test): stage ที่ยังเปิด (NEW/QUALIFIED/PROPOSAL) ย้ายไป stage เปิดอื่นหรือไป LOST ได้, WON มาจาก PROPOSAL เท่านั้น, LOST → NEW ได้ (reopen), WON เป็นสถานะสุดท้าย, LOST ต้องมี `lostReason`
- **Activity เป็น append-only** — ไม่มี endpoint ให้แก้หรือลบ ยกเว้นการ mark TASK ว่าเสร็จ → timeline เชื่อถือได้
- **ไม่ลบ Lead** (ปิดเป็น LOST แทน) และไม่ลบ User (ใช้ `isActive=false`)
- **LINE mapping**: `lineUserId` → Contact (ถ้าไม่มีให้สร้างจาก LINE profile) → lead ที่ยังเปิดและอัปเดตล่าสุด (ถ้าไม่มีให้สร้าง Lead `source=LINE`, `stage=NEW`, ยังไม่มี owner)
- **ไม่มีตาราง AuditLog แยก** — `Activity` + `AiSuggestion` + `Message` (`sentById`, `aiSuggestionId`) ครอบคลุม audit trail ของ MVP แล้ว ส่วน AuditLog กลางเป็น production next step
- **Seed** (faker แบบ fixed seed ให้ได้ข้อมูลเดิมทุกครั้ง): 20 users (1 admin), ~150 companies, 2,000 contacts, 300 active leads + ~150 leads ที่ปิดแล้ว, activities และ LINE messages; รหัสผ่าน demo อ่านจาก env `SEED_DEMO_PASSWORD`

### Flow สำคัญ

**Approve suggestion** (หัวใจของกฎ "AI ห้ามเขียน DB")
1. ใน transaction: `updateMany where { id, status: PENDING }` → `APPROVED` + reviewer ถ้าจำนวนแถวที่เปลี่ยนเป็น 0 → 409 (กัน approve ซ้ำหรือส่งซ้ำ)
2. Apply ตาม type: `QUALIFICATION` → `Lead.score`/`summary`; `NEXT_ACTION` → Activity `TASK`; `LINE_REPLY` → Message `OUTBOUND` `QUEUED` พร้อม `retryKey` ใหม่
3. บันทึก Activity `AI_APPROVED` (metadata: `suggestionId`)
4. ส่ง LINE **หลัง commit** (ไม่ถือ transaction ค้างไว้ระหว่างรอ network) → `SENT` หรือ `FAILED` ถ้า `FAILED` ก็กด retry ได้ด้วย `retryKey` เดิม

**LINE webhook**
1. ใช้ `express.raw()` เฉพาะ route นี้ → verify `X-Line-Signature` (HMAC-SHA256 ด้วย channel secret + `timingSafeEqual`) ถ้าไม่ผ่านตอบ 401 และไม่บันทึกอะไรเลย
2. zod parse → insert `WebhookEvent` ทีละ event (ถ้า `eventId` ชน unique แปลว่าซ้ำ → ข้าม) → ตอบ 200 ทันที
3. ประมวลผลแบบ async: map contact/lead → สร้าง inbound Message → `PROCESSED` → สั่ง AI ร่าง reply (ถ้า AI ล้มก็ไม่กระทบ event)
4. ถ้าประมวลผลล้ม → `FAILED` + ตั้ง `nextRetryAt` (backoff 1/5/15 นาที สูงสุด 5 ครั้ง) → retry worker ใน process (ใช้ได้เพราะรัน instance เดียว; production ควรใช้ queue)
5. LINE outbound: ถ้า error เป็น network/5xx/429 → retry แบบ backoff 3 ครั้ง; ได้ 409 ที่มากับ retry key แปลว่า LINE รับไปแล้ว → ถือว่า `SENT`; 4xx อื่น → `FAILED` ทันที

**AI fallback**: เกิดเมื่อ timeout (~25 วินาที), API error, `stop_reason` เป็น `refusal`/`max_tokens`, `parsed_output` เป็น null, zod ไม่ผ่าน หรือไม่มี API key → ใช้ rule-based fallback แทน (คะแนนคิดจาก stage, ความสดของข้อความล่าสุด, มี company/value หรือไม่; reply ใช้ template สุภาพที่ไม่ผูกมัดเรื่องราคา) บันทึก `source=FALLBACK` + `fallbackReason` และยังต้องผ่าน approve เหมือนเดิม

**Guardrails ของ prompt**: ถือว่าข้อความจากลูกค้าเป็น untrusted data (กัน prompt injection), ห้ามสัญญาราคา/ส่วนลด/timeline ที่ไม่มีใน CRM, ตอบเป็นภาษาเดียวกับลูกค้า, ไม่ส่ง email/phone ของ contact เข้า LLM (PII minimization)

## 3. ลำดับการทำงาน 6 phase (รวมประมาณ 16 ชั่วโมง)

### Phase 1 — Foundation: monorepo + DB + seed (~2 ชม.)
- pnpm workspace, `tsconfig.base.json` (strict + `noUncheckedIndexedAccess`), ESLint (`no-explicit-any: error`), Prettier, Vitest
- `docker-compose.yml` (Postgres 17: dev DB + test DB), `.env.example`, GitHub Actions CI (lint + typecheck + test โดยมี Postgres เป็น service)
- `packages/shared`: enums, `canTransition()` + unit test, zod schemas ชุดแรก
- `apps/api`: Express 5, `env.ts` (validate env ด้วย zod, ผิดแล้วหยุดทำงานทันที), pino, error handler, `GET /api/health`
- Prisma schema ครบทุก model + migration แรก (รวม raw SQL constraints) + seed
- `apps/web`: Next.js App Router skeleton + `rewrites` + หน้าที่แสดงสถานะ health
- อัปเดต `CLAUDE.md`: เพิ่มคำสั่งที่ใช้บ่อย และระบุว่า `skills/crm-copilot` เป็น package
- **รันได้:** `docker compose up -d && pnpm i && pnpm db:migrate && pnpm db:seed && pnpm dev` → เว็บแสดง "API OK / DB OK", `pnpm test` ผ่าน, Prisma Studio เห็นข้อมูล seed

### Phase 2 — Auth + CRM API (~3 ชม.)
- login/logout/me, middleware `requireAuth` / `requireRole`, middleware `validate({ body, params, query })`
- CRUD companies/contacts/leads, search/filter/cursor pagination, `/leads/pipeline`, stage transition service, activities, timeline
- `apps/api/requests.http` สำหรับทดสอบด้วยมือ (ใช้เป็น API notes ไปด้วย)
- **Test ที่โจทย์บังคับ #1 (core CRM flow):** login → สร้าง lead → transition ผิดกติกาได้ 409 → LOST โดยไม่มีเหตุผลได้ 400 → transition ถูกกติกา → timeline มี `STAGE_CHANGE` พร้อม actor
- **รันได้:** เรียกทุก endpoint ผ่าน `requests.http` กับข้อมูล seed ได้ และ `pnpm test` ผ่าน

### Phase 3 — CRM Web UI + deploy ครั้งแรก (~3.5 ชม.) → Part 1 เสร็จ
- Tailwind + shadcn/ui, TanStack Query, react-hook-form + zod resolver (ใช้ schema จาก shared)
- หน้า: `/login`, `/leads` (ตาราง search/filter; มือถือแสดงเป็นการ์ด), `/pipeline` (board แยกตาม stage; ย้าย stage ผ่านเมนู — drag เป็น optional), `/leads/[id]` (profile, เปลี่ยน stage พร้อม dialog กรอก lostReason, timeline, เพิ่ม note/task), `/contacts`, `/companies`
- Deploy บน Railway: Postgres + api + web, รัน `prisma migrate deploy` ทุก release, seed ครั้งเดียว
- **รันได้:** เปิด deployed URL → login ด้วยบัญชี demo → ย้าย stage → refresh และ restart service แล้วข้อมูลยังอยู่

### Phase 4 — AI CRM Copilot + approval flow (~3 ชม.)
- `skills/crm-copilot/`: `SKILL.md` (purpose, inputs, outputs, allowed actions, guardrails, failure behavior, eval cases ≥ 5), `prompts/system.md`, `src/` (`runCrmCopilot()`, Claude provider, mock provider, `fallback.ts`), `evals/` (กรณีตัวอย่าง: lead ร้อน, lead เงียบ 30 วัน, prompt injection ในข้อความ LINE, ลูกค้าถามราคา/ส่วนลด, ข้อมูลไม่ครบ, ข้อความภาษาไทย)
- `apps/api`: context builder (ตัด PII ออก), ai-suggestions endpoints, approve/reject service, logic การ supersede
- ตอน approve `LINE_REPLY` ใน phase นี้ใช้ `MockLineClient` (`LINE_MODE=mock`) → message ขึ้น timeline ด้วยสถานะ `SENT` (mock)
- `apps/web`: AI panel ในหน้า lead — ปุ่ม Ask AI, แสดง summary / score + เหตุผล / next action / draft, ปุ่ม approve (แก้ข้อความได้) และ reject, มี badge "Fallback" และแยกให้เห็นชัดว่าอะไรยังรออนุมัติ อะไรยืนยันแล้ว
- **Test ที่โจทย์บังคับ #2 (AI skill behavior/fallback):** mock provider จำลอง timeout / output ไม่ถูก format / refusal → ต้องได้ suggestion แบบ `FALLBACK`; ก่อน approve ต้องไม่มี Lead/Message เปลี่ยน; approve ซ้ำได้ 409; approve ตัวที่ถูก supersede แล้วได้ 409
- **รันได้:** กด Ask AI บน lead ใดก็ได้ → ได้ suggestion (มาจาก Claude ถ้ามี key ไม่งั้นมาจาก fallback) → approve แล้วเห็น score/task/message ใน timeline

### Phase 5 — LINE OA integration (~3 ชม.)
- สิ่งที่ต้องเตรียมเอง (ทำล่วงหน้าได้ตั้งแต่ Phase 1): LINE OA test account + Messaging API channel → ใส่ channel secret และ access token ใน env ของ Railway เท่านั้น
- webhook route, LINE user → contact/lead mapping, สร้าง inbound Message, ให้ AI ร่าง reply อัตโนมัติ
- `HttpLineClient` (push + `X-Line-Retry-Key` + backoff) ห่อด้วย interface เดียวกับ `MockLineClient`; endpoints ส่ง message และ retry; retry worker; admin endpoints สำหรับ webhook-events
- script `pnpm line:simulate "ข้อความ"` → sign event ปลอมด้วย channel secret แล้ว POST เข้า API ที่รันอยู่ในเครื่อง
- **Test ที่โจทย์บังคับ #3 (LINE webhook security/idempotency):** signature ผิด → 401 และไม่มีแถวใหม่ใน DB; `webhookEventId` ซ้ำ → ได้ WebhookEvent 1 แถวและ Message 1 แถว; ส่งล้มแล้ว retry → ใช้ `retryKey` เดิม; mapping สร้าง contact และ lead แค่ครั้งเดียว
- **รันได้:** ส่งข้อความจากมือถือเข้า OA → ข้อความขึ้นใน timeline ของ lead พร้อม AI draft รออนุมัติ → approve → คำตอบถึงมือถือ

### Phase 6 — Hardening, observability, handover (~1.5 ชม.)
- helmet, CORS allowlist, rate limit (login / webhook / public), ตรวจ cookie flags, สแกนหา secret ในประวัติ git, `pnpm audit`
- Stretch: ฟอร์มหน้าเว็บ `/contact-us` + `POST /api/public/leads`
- Monitoring notes: uptime monitor ยิง `/api/health`, alert เมื่อมี webhook/message ที่ `FAILED`, ติดตาม fallback rate และ latency ของ AI
- README (setup/run/deploy, architecture + data-flow diagram แบบ mermaid, API notes, env, trade-offs, limitations, next steps), เกลา `docs/ai-usage-log.md` (เริ่มบันทึกตั้งแต่ 2026-09-13 และเพิ่มต่อทุก session), วิธีทดสอบ LINE + QR code
- Deploy รอบสุดท้าย + smoke test บน production (ส่วนวิดีโอ walkthrough 3–5 นาทีต้องอัดเอง)
- **รันได้:** clone repo ใหม่แล้วทำตาม README จนรันได้ภายในประมาณ 10 นาที, deployed URL และ LINE QR ใช้งานได้

**เหตุผลของลำดับนี้:** Part 1 (50%) deploy เสร็จตั้งแต่ Phase 3 ถ้าเวลาหมดก็ยังมีส่วนที่มีน้ำหนักมากที่สุดอยู่แล้ว ทำ AI ก่อน LINE เพราะ LINE outbound ต้องใช้ approval flow และ Message pipeline ที่สร้างใน Phase 4 (ทดสอบกับ mock adapter ไปก่อนแล้ว)
